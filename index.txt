    <!--
        ======================================================================
        JAVASCRIPT PURO
        Controle de barra de progresso, navegação móvel, efeito tilt no hero
        e revelação de elementos ao rolamento da tela (Intersection Observer).
        ======================================================================
    -->
    <script>
        document.addEventListener('DOMContentLoaded', () => {

            /*
                --------------------------------------------------------------
                1. BARRA DE PROGRESSO DE ROLAGEM DA PÁGINA
                Calcula a porcentagem lida da página e ajusta a largura da barra.
                --------------------------------------------------------------
            */
            const progressoBarra = document.getElementById('progresso-barra');
            
            window.addEventListener('scroll', () => {
                const alturaTotal = document.documentElement.scrollHeight - window.innerHeight;
                if (alturaTotal > 0) {
                    const progresso = (window.scrollY / alturaTotal) * 100;
                    progressoBarra.style.width = `${progresso}%`;
                }
            });

            /*
                --------------------------------------------------------------
                2. MENU HAMBÚRGUER PARA DISPOSITIVOS MÓVEIS
                Alterna a classe 'open' para exibir/ocultar a lista de navegação.
                --------------------------------------------------------------
            */
            const menuBtn = document.getElementById('menu-btn');
            const navLista = document.getElementById('nav-lista');

            menuBtn.addEventListener('click', () => {
                const aberto = navLista.classList.toggle('open');
                menuBtn.setAttribute('aria-expanded', aberto);
            });

            // Fecha o menu móvel automaticamente ao clicar em qualquer link
            navLista.querySelectorAll('a').forEach(link => {
                link.addEventListener('click', () => {
                    navLista.classList.remove('open');
                    menuBtn.setAttribute('aria-expanded', 'false');
                });
            });

            /*
                --------------------------------------------------------------
                3. INTERAÇÃO DE INCLINAÇÃO SUAVE NO HERO (TILT EFFECT)
                Desativado em telas sensíveis ao toque / dispositivos móveis.
                --------------------------------------------------------------
            */
            const photoCard = document.getElementById('photoCard');
            const bgBlob = document.getElementById('bgBlob');
            const ehDispositivoPonteiro = window.matchMedia('(pointer: fine)').matches;

            if (photoCard && ehDispositivoPonteiro) {
                photoCard.addEventListener('mousemove', (e) => {
                    const rect = photoCard.getBoundingClientRect();
                    const x = e.clientX - rect.left - (rect.width / 2);
                    const y = e.clientY - rect.top - (rect.height / 2);

                    const rotX = (-y / rect.height) * 10;
                    const rotY = (x / rect.width) * 10;

                    photoCard.style.transform = `perspective(600px) rotateX(${rotX}deg) rotateY(${rotY}deg) translateY(-4px)`;
                    if (bgBlob) {
                        bgBlob.style.transform = `translate(${x * 0.05}px, ${y * 0.05}px) scale(1.02)`;
                    }
                });

                photoCard.addEventListener('mouseleave', () => {
                    photoCard.style.transform = 'perspective(600px) rotateX(0deg) rotateY(0deg) translateY(0px)';
                    if (bgBlob) {
                        bgBlob.style.transform = 'translate(0px, 0px) scale(1)';
                    }
                });
            }

            /*
                --------------------------------------------------------------
                4. REVELAÇÃO DE SEÇÕES AO ROLAR (INTERSECTION OBSERVER)
                Respeita as preferências do sistema para redução de movimento.
                --------------------------------------------------------------
            */
            const prefereReducaoMovimento = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

            if (!prefereReducaoMovimento) {
                const elementosReveal = document.querySelectorAll('.reveal');

                const observer = new IntersectionObserver((entries) => {
                    entries.forEach(entry => {
                        if (entry.isIntersecting) {
                            entry.target.classList.add('active');
                        }
                    });
                }, {
                    threshold: 0.12
                });

                elementosReveal.forEach(el => observer.observe(el));
            } else {
                // Se o usuário optar por redução de movimento, mostra tudo imediatamente sem animação
                document.querySelectorAll('.reveal').forEach(el => el.classList.add('active'));
            }
        });
    </script>
